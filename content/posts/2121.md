<p>在我们的工作的项目中，有时候我们需要显示规定尺寸的图片，虽然可以通过css来控制显示大小。但是如果图片过大，会造成加载的延迟，影响网站整体性能。因此，我们需要一个服务器来帮助我们进行图片的裁剪。流程大致是，首先我们传给服务器原图像和裁剪的尺寸，然后服务器进行裁剪，生成对应的裁剪图片，下次我们再访问相同图像和相同的裁剪尺寸的时候，我们就不需要裁剪，直接进行图片的访问就行。</p>
<p></p>
<p>Talk is cheap, show me the code.</p>
<p></p>
<p>&lt;?php</p>
<p></p>
<p></p>
<p>// ①构建图片请求地址比如&nbsp; http://xxx.com/resize.php?site=www&amp;width=300&amp;height=200&amp;mode=2&amp;path=uploadfile/helloworld.png</p>
<p>// ②配置nginx重写规则&nbsp; rewrite /s/(.*)/(\d+)x(\d+)-(\d)/(.*) /s/resize.php?site=$1&amp;width=$2&amp;height=$3&amp;mode=$4&amp;path=$5 last;</p>
<p>//③进行裁剪图片的处理</p>
<p></p>
<p>$path = trim($_GET[&#39;path&#39;]);</p>
<p>$mode = intval($_GET[&#39;mode&#39;]);</p>
<p>$site = trim($_GET[&#39;site&#39;]);</p>
<p>$width = intval($_GET[&#39;width&#39;]);</p>
<p>$height = intval($_GET[&#39;height&#39;]);</p>
<p>$site_list = array(&#39;crop&#39; =&gt; &#39;.&#39;);</p>
<p>$orig_dir = dirname(__FILE__);</p>
<p>if (!array_key_exists($site, $site_list)) {</p>
<p>&nbsp; &nbsp; header(&#39;HTTP/1.1 400 Bad Request&#39;);</p>
<p>&nbsp; &nbsp; exit();</p>
<p>}</p>
<p></p>
<p>if ($mode &gt; 3 || $mode &lt; 0) {</p>
<p>&nbsp; &nbsp; header(&#39;HTTP/1.1 400 Bad Request&#39;);</p>
<p>&nbsp; &nbsp; exit();</p>
<p>}</p>
<p></p>
<p>$orig_file = $site_list[$site] . $path;</p>
<p></p>
<p>if (!file_exists($orig_file)) {</p>
<p>&nbsp; &nbsp; header(&#39;HTTP/1.1 404 Not Found&#39;);</p>
<p>&nbsp; &nbsp; exit();</p>
<p>}</p>
<p></p>
<p>$file_ext = &#39;.&#39; . pathinfo($path, PATHINFO_EXTENSION);</p>
<p>$file_name = basename($path, $file_ext);</p>
<p>$save_path = &quot;{$orig_dir}/{$site}/{$width}x{$height}-{$mode}{$path}&quot;;</p>
<p>$save_dir = dirname($save_path);</p>
<p></p>
<p>if (!file_exists($save_dir)) {</p>
<p>&nbsp; &nbsp; wpx_mkdir($save_dir);</p>
<p>}</p>
<p></p>
<p>$target_width = $width;</p>
<p>$target_height = $height;</p>
<p></p>
<p>$save_image = $save_dir . &#39;/&#39; . $file_name . &#39;.jpg&#39;;</p>
<p>if (file_exists($save_image)) {</p>
<p>&nbsp; &nbsp; header(&#39;Content-Type: image/jpeg&#39;);</p>
<p>&nbsp; &nbsp; header(&#39;Last-Modified: &#39; . gmdate(&#39;D, d M Y H:i:s&#39;) . &#39; GMT&#39;);</p>
<p>&nbsp; &nbsp; echo file_get_contents($save_image);</p>
<p>}</p>
<p></p>
<p>imagecropper2($orig_file, $target_width, $target_height, $save_image);</p>
<p></p>
<p>die;</p>

<p>//原图像对应缩放裁剪，会拉伸图片</p>
<p>function imagecropper2($source_path, $width, $height, $save_image)</p>
<p>{</p>
<p>&nbsp; &nbsp;&nbsp;</p>
<p>&nbsp; &nbsp; //获取原图像$filename的宽度$width_orig和高度$height_orig</p>
<p>&nbsp; &nbsp; $info =&nbsp; getimagesize($source_path);</p>
<p>&nbsp; &nbsp;&nbsp;</p>
<p>&nbsp; &nbsp; $width_orig = $info[0];</p>
<p>&nbsp; &nbsp; $height_orig = $info[1];</p>
<p>&nbsp; &nbsp; $mime = $info[&#39;mime&#39;];</p>
<p>&nbsp; &nbsp;&nbsp;</p>
<p>&nbsp; &nbsp; //根据参数$width和$height值，换算出等比例缩放的高度和宽度</p>
<p>&nbsp; &nbsp; if ($width &amp;&amp; ($width_orig&lt;$height_orig)){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $width = ($height/$height_orig)*$width_orig;</p>
<p>&nbsp; &nbsp; }else{</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $height = ($width / $width_orig)*$height_orig;</p>
<p>&nbsp; &nbsp; }</p>
<p></p>
<p>&nbsp; &nbsp; //将原图缩放到这个新创建的图片资源中</p>
<p>&nbsp; &nbsp; $image_p = imagecreatetruecolor($width, $height);</p>
<p>&nbsp; &nbsp; //获取原图的图像资源</p>
<p>&nbsp; &nbsp; if($mime==&#39;image/jpeg&#39;){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $image = imagecreatefromjpeg($source_path);</p>
<p>&nbsp; &nbsp; }elseif($mime==&#39;image/png&#39;){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $image = imagecreatefrompng($source_path);</p>
<p>&nbsp; &nbsp; }elseif($mime==&#39;image/gif&#39;){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $image = imagecreatefromgif($source_path);</p>
<p>&nbsp; &nbsp; }</p>
<p>&nbsp; &nbsp;&nbsp;</p>
<p></p>
<p>&nbsp; &nbsp; //使用imagecopyresampled()函数进行缩放设置</p>
<p>&nbsp; &nbsp; imagecopyresampled($image_p,$image,0,0,0,0,$width,$height,$width_orig,$height_orig);</p>
<p>&nbsp; &nbsp;</p>
<p>&nbsp; &nbsp; //将缩放后的图片$image_p保存，100(质量最佳，文件最大)</p>
<p>&nbsp; &nbsp; if($mime==&#39;image/jpeg&#39;){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagejpeg($image_p,$save_image);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; header(&#39;Content-Type: image/jpeg&#39;);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagejpeg($image_p);</p>
<p>&nbsp; &nbsp; }elseif($mime==&#39;image/png&#39;){</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagepng($image_p,$save_image);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; header(&#39;Content-Type: image/jpeg&#39;);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagepng($image_p);</p>
<p>&nbsp; &nbsp; }else{</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagegif($image_p,$save_image);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; header(&#39;Content-Type: image/jpeg&#39;);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; imagegif($image_p);</p>
<p>&nbsp; &nbsp; }</p>
<p>&nbsp; &nbsp;&nbsp;</p>
<p>}</p>
<p></p>
<p>//进行比例保存裁剪，会丢失图像部分像素</p>
<p>function imagecropper($source_path, $target_width, $target_height, $save_image)</p>
<p>{</p>
<p>&nbsp; &nbsp; $source_info = getimagesize($source_path);</p>
<p>&nbsp; &nbsp; $source_width = $source_info[0];</p>
<p>&nbsp; &nbsp; $source_height = $source_info[1];</p>
<p>&nbsp; &nbsp; $source_mime = $source_info[&#39;mime&#39;];</p>
<p>&nbsp; &nbsp; $source_ratio = $source_height / $source_width;</p>
<p>&nbsp; &nbsp; $target_ratio = $target_height / $target_width;</p>
<p></p>
<p>&nbsp; &nbsp; // 源图过高</p>
<p>&nbsp; &nbsp; if ($source_ratio &gt; $target_ratio) {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_width = $source_width;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_height = $source_width * $target_ratio;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_x = 0;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_y = ($source_height &#8211; $cropped_height) / 2;</p>
<p>&nbsp; &nbsp; }</p>
<p>&nbsp; &nbsp; // 源图过宽</p>
<p>&nbsp; &nbsp; elseif ($source_ratio &lt; $target_ratio) {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_width = $source_height / $target_ratio;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_height = $source_height;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_x = ($source_width &#8211; $cropped_width) / 2;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_y = 0;</p>
<p>&nbsp; &nbsp; }</p>
<p>&nbsp; &nbsp; // 源图适中</p>
<p>&nbsp; &nbsp; else {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_width = $source_width;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $cropped_height = $source_height;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_x = 0;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; $source_y = 0;</p>
<p>&nbsp; &nbsp; }</p>
<p></p>
<p>&nbsp; &nbsp; switch ($source_mime) {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; case &#39;image/gif&#39;:</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; $source_image = imagecreatefromgif($source_path);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; break;</p>
<p></p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; case &#39;image/jpeg&#39;:</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; $source_image = imagecreatefromjpeg($source_path);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; break;</p>
<p></p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; case &#39;image/png&#39;:</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; $source_image = imagecreatefrompng($source_path);</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; break;</p>
<p></p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; default:</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; return false;</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; break;</p>
<p>&nbsp; &nbsp; }</p>
<p></p>
<p>&nbsp; &nbsp; $target_image = imagecreatetruecolor($target_width, $target_height);</p>
<p>&nbsp; &nbsp; $cropped_image = imagecreatetruecolor($cropped_width, $cropped_height);</p>
<p></p>
<p>&nbsp; &nbsp; // 裁剪</p>
<p>&nbsp; &nbsp; $bool = imagecopy($cropped_image, $source_image, 0, 0, $source_x, $source_y, $cropped_width, $cropped_height);</p>
<p></p>
<p>&nbsp; &nbsp; // 缩放</p>
<p>&nbsp; &nbsp; $bool = imagecopyresampled($target_image, $cropped_image, 0, 0, 0, 0, $target_width, $target_height, $cropped_width, $cropped_height);</p>
<p>&nbsp; &nbsp; imagejpeg($target_image, $save_image);</p>
<p>&nbsp; &nbsp; header(&#39;Content-Type: image/jpeg&#39;);</p>
<p></p>
<p>&nbsp; &nbsp; imagejpeg($target_image);</p>
<p>&nbsp; &nbsp; imagedestroy($source_image);</p>
<p>&nbsp; &nbsp; imagedestroy($target_image);</p>
<p>&nbsp; &nbsp; imagedestroy($cropped_image);</p>
<p>}</p>

<p>// 循环生成目录</p>
<p>function wpx_mkdir($dir, $mode = 0777)</p>
<p>{</p>
<p>&nbsp; &nbsp; if (is_dir($dir) || @mkdir($dir, $mode)) {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; return true;</p>
<p>&nbsp; &nbsp; }</p>
<p></p>
<p>&nbsp; &nbsp; if (!wpx_mkdir(dirname($dir), $mode)) {</p>
<p>&nbsp; &nbsp; &nbsp; &nbsp; return false;</p>
<p>&nbsp; &nbsp; }</p>
<p></p>
<p>&nbsp; &nbsp; return @mkdir($dir, $mode);</p>
<p>}</p>
<p></p>
<p>通过上面的处理，我们就将图片按照我们设置的尺寸进行了裁剪。我们还可以定期对裁剪图片进行清理，这样就不需要占用太多服务器空间。只有经常访问的图片才会一直保存。</p>
