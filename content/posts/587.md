<p>下面介绍 3种php远程获取图片的方法，希望对大家有帮助。</p>
<p><span style="color: rgb(80, 80, 80); line-height: 28px; font-family: 宋体, &quot;Arial Narrow&quot;, arial, serif; font-size: 14px; font-style: normal; font-weight: normal; background-color: rgb(255, 255, 255);"><strong>1：file_get_contents,file_put_contents方式</strong></span> </p>
<pre class="prettyprint lang-js">function&nbsp;download_remote_file($file_url,&nbsp;$save_to){

		$content&nbsp;=&nbsp;file_get_contents($file_url);

		file_put_contents($save_to,&nbsp;$content);

}</pre>
<p><span style="color: rgb(80, 80, 80); line-height: 28px; font-family: 宋体, &quot;Arial Narrow&quot;, arial, serif; font-size: 14px; font-style: normal; font-weight: normal; background-color: rgb(255, 255, 255);"><strong>用法：</strong></span> </p>
<pre class="prettyprint lang-js">download_remote_file_with_curl(&#39;http://www.54ux.com/wp-content/themes/d-simple/img/thumbnail.jpg&#39;,&nbsp;realpath(&quot;./downloads&quot;)&nbsp;.&nbsp;&#39;/file.jpg&#39;);</pre>
<p><strong>2：curl方式</strong> </p>
<pre class="prettyprint lang-js">function&nbsp;download_remote_file_with_curl($file_url,&nbsp;$save_to)

{

&nbsp;&nbsp;&nbsp;$ch&nbsp;=&nbsp;curl_init();

　&nbsp;curl_setopt($ch,&nbsp;CURLOPT_POST,&nbsp;0);

　&nbsp;curl_setopt($ch,CURLOPT_URL,$file_url);

　&nbsp;curl_setopt($ch,&nbsp;CURLOPT_RETURNTRANSFER,&nbsp;1);

&nbsp;&nbsp;&nbsp;$file_content&nbsp;=&nbsp;curl_exec($ch);

&nbsp;&nbsp;&nbsp;curl_close($ch);

&nbsp;&nbsp;&nbsp;$downloaded_file&nbsp;=&nbsp;fopen($save_to,&nbsp;&#39;w&#39;);

&nbsp;&nbsp;&nbsp;fwrite($downloaded_file,&nbsp;$file_content);

&nbsp;&nbsp;&nbsp;fclose($downloaded_file);

}</pre>
<p><strong>用法：</strong> </p>
<pre class="prettyprint lang-php">download_remote_file_with_curl(&#39;http://www.54ux.com/wp-content/themes/d-simple/img/thumbnail.jpg&#39;,&nbsp;realpath(&quot;./downloads&quot;)&nbsp;.&nbsp;&#39;/file.jpg&#39;);</pre>
<p><strong>3：fopen方法</strong> </p>
<pre class="prettyprint lang-js">function&nbsp;download_remote_file_with_fopen($file_url,&nbsp;$save_to){

&nbsp;&nbsp;&nbsp;$in=&nbsp;fopen($file_url,&nbsp;&quot;rb&quot;);

　&nbsp;$out=fopen($save_to,&nbsp;&quot;wb&quot;);

&nbsp;&nbsp;&nbsp;while&nbsp;($chunk&nbsp;=&nbsp;fread($in,8192)){

&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;fwrite($out,&nbsp;$chunk,&nbsp;8192);

&nbsp;&nbsp;&nbsp;}

&nbsp;&nbsp;&nbsp;fclose($in);

&nbsp;&nbsp;&nbsp;fclose($out);

}</pre>
<p><strong>用法：</strong> </p>
<pre class="prettyprint lang-php">download_remote_file_with_fopen(&#39;http://www.54ux.com/wp-content/themes/d-simple/img/thumbnail.jpg&#39;,&nbsp;realpath(&quot;./downloads&quot;)&nbsp;.&nbsp;&#39;/file.jpg&#39;);</pre>
