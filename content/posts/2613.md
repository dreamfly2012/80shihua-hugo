<p>使用管理员执行下面的挂在虚拟内存命令</p>
<pre><code class="language-shell">#Check free memory before
free -m

mkdir -p /var/_swap_
cd /var/_swap_
#Here, 1M * 2000 ~= 2GB of swap memory.  Feel free to add MORE
dd if=/dev/zero of=swapfile bs=1M count=2000
mkswap swapfile
swapon swapfile
chmod 600 swapfile
#Automatically mount this swap partition at startup
echo &quot;/var/_swap_/swapfile none swap sw 0 0&quot; &gt;&gt; /etc/fstab

#Check free memory after
free -m</code></pre>
