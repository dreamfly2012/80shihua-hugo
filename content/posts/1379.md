<div><span style="color: #f92672;">&lt;?php</span></div>
<div><span style="color: #75715e;">/**</span></div>
<div><span style="color: #75715e;"> * 并归排序</span></div>
<div><span style="color: #75715e;"> * 思想:将数据排序，一个小的，一个大的，这样遍历数据得时候，这样第一次遍历得时候可以得到一个(小，大),(小,大)数组</span></div>
<div><span style="color: #75715e;"> * 然后通过递归合并，(小1,大1),(小2,大2)每次取出2个数组的第一个数，比较，哪个小，放到一个新的数据tmp里面，然后小的数组current指针后移，继续比较，如果大1&lt;小2，将大1放到</span></div>
<div><span style="color: #75715e;"> * tmp里面，然后将剩余的小2，大2依次放入数组，就是两个数组合并排序完了，即tmp = array(小1,大1,小2,大2)</span></div>
<div><span style="color: #75715e;"> * </span></div>
<div><span style="color: #75715e;"> * 如果刚才大1&gt;小2,则先放入小2,然后（小2，大2）的数组current后移，然后比较大1和大2，如果大1&gt;大2，则大2先放入tmp数组，剩下的大1，放入数组，就是(小1，小2，大2，大1)升序的排好数组</span></div>
<div><span style="color: #75715e;"> */</span></div>
<div><span style="color: #75715e;">//测试类</span></div>
<div><span style="color: #66d9ef;font-style: italic;">class</span> <span style="color: rgb(166, 226, 46); text-decoration-line: underline;">MergeTest</span></div>
<div>{</div>
<div><span style="color: #f92672;">private</span> $data;</div>
<div><span style="color: #f92672;">public</span> <span style="color: #66d9ef;font-style: italic;">function</span> <span style="color: #66d9ef;">__construct</span>($data)</div>
<div>{</div>
<div><span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span><span style="color: #a6e22e;">init</span>($data);</div>
<div>}</div>
<p></p>
<div><span style="color: #75715e;">//执行</span></div>
<div><span style="color: #f92672;">public</span> <span style="color: #66d9ef;font-style: italic;">function</span> <span style="color: #a6e22e;">run</span>()</div>
<div>{</div>
<div><span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span><span style="color: #a6e22e;">merge</span>();</div>
<div><span style="color: #66d9ef;">print_r</span>(<span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data);</div>
<div>}</div>
<div><span style="color: #75715e;">//初始化并归数组,相邻两个元素合并为一个数组(小的在前面)</span></div>
<div><span style="color: #f92672;">private</span> <span style="color: #66d9ef;font-style: italic;">function</span> <span style="color: #a6e22e;">init</span>($data)</div>
<div>{</div>
<div>$tmp <span style="color: #f92672;">=</span> [];</div>
<div>$num <span style="color: #f92672;">=</span> <span style="color: #66d9ef;">count</span>($data);</div>
<div><span style="color: #f92672;">if</span> ($num <span style="color: #f92672;">%</span> <span style="color: #ae81ff;">2</span> <span style="color: #f92672;">==</span> <span style="color: #ae81ff;">0</span>) {</div>
<div>$items <span style="color: #f92672;">=</span> $num <span style="color: #f92672;">/</span> <span style="color: #ae81ff;">2</span>;</div>
<div>} <span style="color: #f92672;">else</span> {</div>
<div>$items <span style="color: #f92672;">=</span> ($num <span style="color: #f92672;">&#8211;</span> <span style="color: #ae81ff;">1</span>) <span style="color: #f92672;">/</span> <span style="color: #ae81ff;">2</span>;</div>
<div>}</div>
<p></p>
<div><span style="color: #f92672;">for</span> ($i <span style="color: #f92672;">=</span> <span style="color: #ae81ff;">0</span>; $i <span style="color: #f92672;">&lt;</span> $items; $i<span style="color: #f92672;">++</span>) {</div>
<div><span style="color: #f92672;">if</span> ($data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i] <span style="color: #f92672;">&lt;</span> $data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i <span style="color: #f92672;">+</span> <span style="color: #ae81ff;">1</span>]) {</div>
<div><span style="color: #66d9ef;">array_push</span>($tmp, [$data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i], $data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i <span style="color: #f92672;">+</span> <span style="color: #ae81ff;">1</span>]]);</div>
<div>} <span style="color: #f92672;">else</span> {</div>
<div><span style="color: #66d9ef;">array_push</span>($tmp, [$data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i <span style="color: #f92672;">+</span> <span style="color: #ae81ff;">1</span>], $data[<span style="color: #ae81ff;">2</span> <span style="color: #f92672;">*</span> $i]]);</div>
<div>}</div>
<p></p>
<div>}</div>
<div><span style="color: #f92672;">if</span> ($num <span style="color: #f92672;">%</span> <span style="color: #ae81ff;">2</span> <span style="color: #f92672;">!=</span> <span style="color: #ae81ff;">0</span>) {</div>
<div><span style="color: #66d9ef;">array_push</span>($tmp, [$data[$num <span style="color: #f92672;">&#8211;</span> <span style="color: #ae81ff;">1</span>]]);</div>
<div>}</div>
<p></p>
<div><span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data <span style="color: #f92672;">=</span> $tmp;</div>
<div>}</div>
<p></p>
<div><span style="color: #75715e;">//并归递归调用</span></div>
<div><span style="color: #f92672;">private</span> <span style="color: #66d9ef;font-style: italic;">function</span> <span style="color: #a6e22e;">merge</span>()</div>
<div>{</div>
<div>$re <span style="color: #f92672;">=</span> [];</div>
<div>$size <span style="color: #f92672;">=</span> <span style="color: #66d9ef;">count</span>(<span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data);</div>
<div><span style="color: #f92672;">if</span> ($size <span style="color: #f92672;">!=</span> <span style="color: #ae81ff;">1</span>) <span style="color: #75715e;">//没有完成并归,继续</span></div>
<div>{</div>
<div><span style="color: #f92672;">for</span> ($i <span style="color: #f92672;">=</span> <span style="color: #ae81ff;">0</span>; $i <span style="color: #f92672;">&lt;</span> $size <span style="color: #f92672;">&#8211;</span> <span style="color: #ae81ff;">1</span>; $i <span style="color: #f92672;">+=</span> <span style="color: #ae81ff;">2</span>) {</div>
<div>$re[] <span style="color: #f92672;">=</span> <span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span><span style="color: #a6e22e;">doubleMerge</span>(<span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data[$i], <span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data[$i <span style="color: #f92672;">+</span> <span style="color: #ae81ff;">1</span>]);</div>
<div>}</div>
<p></p>
<div><span style="color: #f92672;">if</span> ($size <span style="color: #f92672;">%</span> <span style="color: #ae81ff;">2</span> <span style="color: #f92672;">!=</span> <span style="color: #ae81ff;">0</span>) {</div>
<div>$re[] <span style="color: #f92672;">=</span> <span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data[$size <span style="color: #f92672;">&#8211;</span> <span style="color: #ae81ff;">1</span>];</div>
<div>}</div>
<p></p>
<div><span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span>data <span style="color: #f92672;">=</span> $re;</div>
<div><span style="color: #fd971f;">$this</span><span style="color: #f92672;">-&gt;</span><span style="color: #a6e22e;">merge</span>();</div>
<div>}</div>
<div>}</div>
<p></p>
<div><span style="color: #75715e;">//合并两个有序序列</span></div>
<div><span style="color: #f92672;">private</span> <span style="color: #66d9ef;font-style: italic;">function</span> <span style="color: #a6e22e;">doubleMerge</span>($a, $b)</div>
<div>{</div>
<div>$re <span style="color: #f92672;">=</span> [];</div>
<div>$i <span style="color: #f92672;">=</span> $j <span style="color: #f92672;">=</span> <span style="color: #ae81ff;">0</span>;</div>
<div>$s1 <span style="color: #f92672;">=</span> <span style="color: #66d9ef;">count</span>($a);</div>
<div>$s2 <span style="color: #f92672;">=</span> <span style="color: #66d9ef;">count</span>($b);</div>
<p></p>
<div><span style="color: #f92672;">while</span> (<span style="color: #ae81ff;">true</span>) {</div>
<div><span style="color: #f92672;">if</span> ($a[$i] <span style="color: #f92672;">&lt;=</span> $b[$j]) {</div>
<div><span style="color: #66d9ef;">array_push</span>($re, $a[$i]);</div>
<div>$i<span style="color: #f92672;">++</span>;</div>
<div>} <span style="color: #f92672;">else</span> {</div>
<div><span style="color: #66d9ef;">array_push</span>($re, $b[$j]);</div>
<div>$j<span style="color: #f92672;">++</span>;</div>
<div>}</div>
<p></p>
<div><span style="color: #f92672;">if</span> ($i <span style="color: #f92672;">==</span> $s1) {</div>
<div><span style="color: #f92672;">return</span> <span style="color: #66d9ef;">array_merge</span>($re, <span style="color: #66d9ef;">array_slice</span>($b, $j));</div>
<div>}</div>
<p></p>
<div><span style="color: #f92672;">if</span> ($j <span style="color: #f92672;">==</span> $s2) {</div>
<div><span style="color: #f92672;">return</span> <span style="color: #66d9ef;">array_merge</span>($re, <span style="color: #66d9ef;">array_slice</span>($a, $i));</div>
<div>}</div>
<p></p>
<div>}</div>
<p></p>
<div>}</div>
<div>}</div>
<p></p>
<div>$a <span style="color: #f92672;">=</span> <span style="color: #f92672;">new</span> <span style="color: #66d9ef;font-style: italic;">MergeTest</span>([<span style="color: #ae81ff;">3</span>,<span style="color: #ae81ff;">8</span>,<span style="color: #ae81ff;">8</span>,<span style="color: #ae81ff;">19</span>, <span style="color: #ae81ff;">1</span>, <span style="color: #ae81ff;">9</span>, <span style="color: #ae81ff;">2</span>, <span style="color: #ae81ff;">7</span>, <span style="color: #ae81ff;">6</span>, <span style="color: #ae81ff;">11</span>,<span style="color: #ae81ff;">19</span>, <span style="color: #ae81ff;">5</span>, <span style="color: #ae81ff;">4</span>, <span style="color: #ae81ff;">13</span>]);</div>
