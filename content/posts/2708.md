bat脚本

```
<code>@echo on
echo "请注意你的杀毒软件提示，一定要允许"
@echo  ########################################
@xcopy C:\Windows\system32\drivers\etc\hosts C:\Windows\system32\drivers\etc\hosts.bak\ /d /c /i /y
@echo  ########################################
@echo  hosts文件备份完毕，开始修改hosts文件

@echo.>>C:\Windows\System32\drivers\etc\hosts
@echo 140.82.112.4 github.com>>C:\Windows\System32\drivers\etc\hosts
@echo 140.82.113.3 gist.github.com>>C:\Windows\System32\drivers\etc\hosts
@echo 199.232.69.194 github.global.ssl.fastly.net>>C:\Windows\System32\drivers\etc\hosts
@echo 185.199.111.153 assets-cdn.github.com>>C:\Windows\System32\drivers\etc\hosts
@echo 199.232.68.133 raw.githubusercontent.com>>C:\Windows\System32\drivers\etc\hosts

echo   "hosts文件修改完成"
@ipconfig /flushdns
@echo   "刷新DNS完成"
pause</code>
```
