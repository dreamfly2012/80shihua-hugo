<h2>操作步骤</h2>
<ol>
<li>安装sql驱动，go get github.com/Go-SQL-Driver/MySQL</li>
<li>引入sql驱动</li>
<li>声明全局变量，var db *sql.DB</li>
<li>连接数据库sql.Open(&quot;mysql&quot;,&quot;root:root@tcp(127.0.0.1:3306)/test_new&quot;);</li>
<li>进行数据的查询 db.Query(&quot;select id,title from news&quot;)</li>
</ol>
<p>具体代码如下：</p>
<pre><code class="language-go">func handleDb(w http.ResponseWriter, r *http.Request) {
    db, err := sql.Open(&quot;mysql&quot;, &quot;root:root@tcp(127.0.0.1:3306)/test_new&quot;)

    if err != nil {
        panic(err)
    }
    rows, err := db.Query(&quot;select id,title from news&quot;)
    if err != nil {
        panic(err)
    }

    var (
        id string
        title string
    )

    var news []string
    for rows.Next() {
        rows.Scan(&amp;id, &amp;title)
        news = append(news, title)
    }
    templates.ExecuteTemplate(w, &quot;db.html&quot;, news)
}</code></pre>
