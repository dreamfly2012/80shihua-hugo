```php
<?php
$ch = curl_init();
$url = "http://www.seouldrama.org/process/entry/RecommEntryData.ashx";
$header = array(
'CLIENT-IP:101.68.44.61',
'X-FORWARDED-FOR:101.68.44.61',
);
$post_data = array ("entryIdx" => "786");
//curl_setopt($ch, CURLOPT_PROXY, "http://223.99.189.102:8090");
curl_setopt($ch, CURLOPT_PROXY, "221.181.228.214"); //代理服务器地址
curl_setopt($ch, CURLOPT_PROXYPORT, 80); //代理服务器端口
curl_setopt($ch, CURLOPT_POST, 1);
curl_setopt($ch, CURLOPT_USERAGENT, "Mozilla/5.0 (Windows NT 6.1) AppleWebKit/536.11 (KHTML, like Gecko) Chrome/20.0.1132.57 Safari/536.11");
curl_setopt($ch, CURLOPT_POSTFIELDS, $post_data);
curl_setopt($ch, CURLOPT_URL, $url);
//curl_setopt($ch, CURLOPT_HTTPHEADER, $header);
curl_setopt($ch, CURLOPT_RETURNTRANSFER,true);
$page_content = curl_exec($ch);
curl_close($ch);
echo $page_content;
?>
```
