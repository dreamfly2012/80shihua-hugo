<h2>匿名函数的引入</h2>
<pre><code class="language-php">&lt;?php
$message = &#039;hello&#039;;

// 没有 &quot;use&quot;
$example = function () {
    var_dump($message);
};
echo $example();

// 继承 $message
$example = function () use ($message) {
    var_dump($message);
};
echo $example();

// Inherited variable&#039;s value is from when the function
// is defined, not when called
$message = &#039;world&#039;;
echo $example();

// Reset message
$message = &#039;hello&#039;;

// Inherit by-reference
$example = function () use (&amp;$message) {
    var_dump($message);
};
echo $example();

// The changed value in the parent scope
// is reflected inside the function call
$message = &#039;world&#039;;
echo $example();

// Closures can also accept regular arguments
$example = function ($arg) use ($message) {
    var_dump($arg . &#039; &#039; . $message);
};
$example(&quot;hello&quot;);

$course = &#039;数学&#039;;
 $func = function ($value,$key)use ($course){
     echo $value.$course.$key;
 };
 $arr = [
     1=&gt;1,
     2=&gt;2,
     3=&gt;3
 ];
 array_walk($arr,$func);
</code></pre>
<p>结果：</p>
<pre><code class="language-shell">Notice: Undefined variable: message in /example.php on line 6
NULL

string(5) &quot;hello&quot;
string(5) &quot;hello&quot;
string(5) &quot;world&quot;
string(11) &quot;hello world&quot;
输出&quot;1数学12数学23数学3&quot;</code></pre>

<h2>Void 函数</h2>
<pre><code class="language-php">&lt;?php
function swap(&amp;$left, &amp;$right) : void
{
    if ($left === $right) {
        return;
    }

    $tmp = $left;
    $left = $right;
    $right = $tmp;
}

$a = 1;
$b = 2;
var_dump(swap($a, $b), $a, $b);</code></pre>

[$id1, $name1] = $data[0];
