```
$handler = opendir('./templates/zh/sj/');//当前目录中的文件夹下的文件夹
$folder_list = array();
while( ($filename = readdir($handler)) !== false ) {
    if($filename != "." && $filename != ".."){
        array_push($folder_list,$filename);
    }
}
closedir($handler);
```
