$arr=array(1,43,54,62,21,66,32,78,36,76,39);



function&nbsp;maopao($arr)

{

$len=count($arr);

//设置一个空数组&nbsp;用来接收冒出来的泡

//该层循环控制&nbsp;需要冒泡的轮数

for($i=1;$i&lt;$len;$i++)

{&nbsp;//该层循环用来控制每轮&nbsp;冒出一个数&nbsp;需要比较的次数

for($k=0;$k&lt;$len-$i;$k++)

{

if($arr[$k]&gt;$arr[$k+1])

{

$tmp=$arr[$k+1];

$arr[$k+1]=$arr[$k];

$arr[$k]=$tmp;

}

}

}

return&nbsp;$arr;

}
